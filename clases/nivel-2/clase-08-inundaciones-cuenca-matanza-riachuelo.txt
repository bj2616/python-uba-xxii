#hoy analizamos datos de precipitacion en la cuenca Matanza-Riachuelo
#y vemos si superan un umbral de riesgo de inundacion

import csv

archivo = open("precipitacion_cuenca.csv")
lector = csv.reader(archivo)
next(lector)

dias_riesgo = 0
umbral = 50  # milimetros, valor de ejemplo

for fila in lector:
    mm = float(fila[0])
    if mm > umbral:
        dias_riesgo = dias_riesgo + 1

archivo.close()

print("cantidad de dias con riesgo de inundacion:", dias_riesgo)
