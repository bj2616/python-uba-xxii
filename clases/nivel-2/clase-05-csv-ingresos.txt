#ingresos.csv

import csv

archivo = open("ingresos.csv")
lector = csv.reader(archivo)
next(lector)

ingresos = []
for fila in lector:
    ingresos.append(float(fila[0]))

archivo.close()

total = sum(ingresos)
promedio = total / len(ingresos)

print("ingreso total:", total)
print("ingreso promedio:", promedio)
